<script>
function getPMIRate(credit, ltv) {
    const table = {
        "760": { "95": 0.0030, "90": 0.0020, "85": 0.0010 },
        "740": { "95": 0.0035, "90": 0.0025, "85": 0.0015 },
        "720": { "95": 0.0040, "90": 0.0030, "85": 0.0020 },
        "700": { "95": 0.0045, "90": 0.0035, "85": 0.0025 },
        "680": { "95": 0.0050, "90": 0.0040, "85": 0.0030 },
        "660": { "95": 0.0060, "90": 0.0050, "85": 0.0040 },
        "640": { "95": 0.0070, "90": 0.0060, "85": 0.0050 },
        "620": { "95": 0.0090, "90": 0.0080, "85": 0.0070 }
    };
    if (ltv > 0.95) return table[credit]["95"];
    if (ltv > 0.90) return table[credit]["90"];
    return table[credit]["85"];
}

function calculatePITI() {
    const loanType = document.getElementById("loanType").value;
    const credit = document.getElementById("credit").value;
    const price = +document.getElementById("price").value;
    const down = +document.getElementById("down").value;
    const rate = +document.getElementById("rate").value / 100 / 12;
    const term = +document.getElementById("term").value * 12;
    const tax = +document.getElementById("tax").value / 12;
    const insurance = +document.getElementById("insurance").value / 12;
    const hoa = +document.getElementById("hoa").value;

    let loan = price - down;

    if (loanType === "fha") {
        loan *= 1.0175;
    }

    const PI = loan * (rate * Math.pow(1 + rate, term)) / (Math.pow(1 + rate, term) - 1);

    let PMI = 0;
    let MIP = 0;

    const ltv = loan / price;

    if (loanType === "conv" && ltv > 0.80) {
        const pmiRate = getPMIRate(credit, ltv);
        PMI = (loan * pmiRate) / 12;
    }

    if (loanType === "fha") {
        const mipRate = loan <= 726200 ? 0.0055 : 0.0075;
        MIP = (loan * mipRate) / 12;
    }

    const total = PI + tax + insurance + hoa + PMI + MIP;

    document.getElementById("output").innerHTML =
        "$" + total.toFixed(2).toLocaleString() + " / month";

    drawEquityChart(price, loan, rate, term);
}

function drawEquityChart(price, loan, rate, term) {
    const labels = [];
    const equity = [];
    let balance = loan;

    for (let i = 1; i <= term; i++) {
        const interest = balance * rate;
        const PI = loan * (rate * Math.pow(1 + rate, term)) / (Math.pow(1 + rate, term) - 1);
        const principal = PI - interest;
        balance -= principal;

        if (i % 12 === 0) {
            labels.push("Year " + (i / 12));
            equity.push(price - balance);
        }
    }

    if (window.equityChart) window.equityChart.destroy();

    const ctx = document.getElementById("equityChart");
    window.equityChart = new Chart(ctx, {
        type: "line",
        data: {
            labels: labels,
            datasets: [{
                label: "Equity Over Time",
                data: equity,
                borderColor: "#2e7d32",
                borderWidth: 3,
                fill: false
            }]
        }
    });
}

document.querySelectorAll("input, select").forEach(el => {
    el.addEventListener("input", calculatePITI);
});

calculatePITI();
