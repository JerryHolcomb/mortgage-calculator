function updateDownLabel() {
    document.getElementById("downLabel").innerText =
        document.getElementById("downPercent").value + "%";
}

function calculate() {
    let price = parseFloat(document.getElementById("price").value);
    let downPercent = parseFloat(document.getElementById("downPercent").value);
    let down = price * (downPercent / 100);

    let rate = parseFloat(document.getElementById("rate").value) / 100 / 12;
    let years = parseFloat(document.getElementById("years").value) * 12;
    let taxes = parseFloat(document.getElementById("taxes").value) / 12;
    let insurance = parseFloat(document.getElementById("insurance").value) / 12;

    let loanAmount = price - down;

    let PI = (loanAmount * rate) / (1 - Math.pow(1 + rate, -years));
    PI = PI.toFixed(2);

    let total = (parseFloat(PI) + taxes + insurance).toFixed(2);

    document.getElementById("pi").innerText = PI;
    document.getElementById("tax").innerText = taxes.toFixed(2);
    document.getElementById("ins").innerText = insurance.toFixed(2);
    document.getElementById("total").innerText = total;

    document.getElementById("results").style.display = "block";
}
